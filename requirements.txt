# StegApp — Python dependencies for Render (and any plain pip install).
#
# Python: 3.11+ (tested on 3.13). On Render, pin it with the PYTHON_VERSION
#         environment variable, e.g. PYTHON_VERSION=3.13.7
#
# Render settings (Web Service, Runtime: Python 3, Root Directory: blank):
#   Build Command:  pip install -r requirements.txt
#   Start Command:  uvicorn app:app --app-dir backend --host 0.0.0.0 --port $PORT
#
# Environment variables (optional, see backend/README.md):
#   STEGAPP_API_KEY       If set, /api/crack requires this exact X-API-Key.
#   STEGAPP_WORK_DIR      Scratch dir for uploads (default backend/work).
#   STEGAPP_WORDLIST_DIR  Extra directory to search for wordlists.
#
# NOT installable via pip: the app shells out to the Linux binaries `steghide`
# and `stegseek`. Render's native Python runtime can't apt-install them, so
# embed/extract/detect/crack will return HTTP 503 there. To get them, deploy
# on Render with Runtime: Docker using docker/Dockerfile instead.
#
# Versions below are the ones verified locally; keep in sync with
# backend/requirements.txt (used by the Docker build).

fastapi==0.162.0             # web framework / API routes + static frontend
starlette==1.10.0            # ASGI toolkit under FastAPI (StaticFiles)
pydantic==2.14.2             # FastAPI request/response models
uvicorn[standard]==0.54.0    # ASGI server that runs the app
python-multipart==0.0.33     # required by FastAPI for file uploads / form fields
